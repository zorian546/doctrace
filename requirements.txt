# Full runtime: the light dependencies plus the ML stack.
#
# Split on purpose. requirements-base.txt holds everything that is not a heavy ML
# runtime, and the test suite needs only that (see requirements-dev.txt), because the
# torch imports in doctrace/ are deferred to first use.

-r requirements-base.txt

# doctrace/answering/local_model.py (Qwen2.5-3B-Instruct) imports transformers directly,
# so it is pinned here rather than left to whatever sentence-transformers pulls in.
# torch is deliberately absent: this project needs a PyTorch NIGHTLY cu128 build for
# Blackwell/sm_120 GPUs, which the default PyPI index does not carry. The Dockerfile
# installs it as a separate step; a local (non-Docker) setup should run the same command.
transformers==5.5.0
sentence-transformers            # encoder and cross-encoder
# local_model.py loads with device_map=..., which transformers only allows when
# `accelerate` is installed. It was missing until a clean Docker build exposed it; on a
# dev machine another package had been pulling it in silently (see ENGINEERING_NOTES.md).
accelerate==1.14.0

# Evaluation.
# ragas is pinned on purpose. Unpinned, a clean install resolves 0.4.3, which crashes at
# import: `from ragas.llms.base import BaseRagasLLM` (used by
# doctrace/audit/ragas_bridge.py) raises ModuleNotFoundError for
# langchain_community.chat_models.vertexai. That was reproduced in a real container
# build, not only locally. LocalRagasJudge also targets the 0.1.21 BaseRagasLLM
# interface (abstract generate_text/agenerate_text), which 0.4.3 never exposes because
# of the crash. Before bumping this, confirm scripts/bench_ragas.py runs end to end; a
# clean `import ragas` is not enough.
ragas==0.1.21

# Versions taken from `pip show` inside a container built against ragas==0.1.21 (never
# against 0.4.3, which does not install cleanly). Pinned so a later install does not
# drift: ragas 0.1.21's own dependency ranges are wide.
pydantic==2.13.4
langchain-core==0.2.43
datasets==5.0.1
numpy==1.26.4
